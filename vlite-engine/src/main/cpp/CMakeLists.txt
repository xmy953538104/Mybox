cmake_minimum_required(VERSION 3.22)
project(mybox_vlite LANGUAGES CXX ASM)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_library(mybox-vlite-core STATIC path_rules.cpp native_io.cpp syscall_gate.S)
set_target_properties(mybox-vlite-core PROPERTIES POSITION_INDEPENDENT_CODE ON)
target_compile_definitions(mybox-vlite-core PRIVATE _GNU_SOURCE)
target_compile_options(mybox-vlite-core PRIVATE -Wall -Wextra -Werror -fno-exceptions -fno-rtti)
if(ANDROID)
    if(NOT ANDROID_ABI STREQUAL "arm64-v8a" AND NOT ANDROID_ABI STREQUAL "x86_64")
        message(FATAL_ERROR "Native I/O supports arm64-v8a and x86_64 only")
    endif()
    add_library(mybox-vlite SHARED native_bridge.cpp native_uid.cpp)
    target_link_libraries(mybox-vlite PRIVATE mybox-vlite-core dl)
    target_link_options(mybox-vlite PRIVATE -Wl,-z,now -Wl,-z,relro -Wl,-z,max-page-size=16384)
else()
    enable_testing()
    add_executable(mybox-vlite-native-test tests/native_core_test.cpp)
    target_link_libraries(mybox-vlite-native-test PRIVATE mybox-vlite-core pthread)
    target_link_options(mybox-vlite-native-test PRIVATE -Wl,-z,now)
    add_test(NAME native-io COMMAND mybox-vlite-native-test)
endif()
