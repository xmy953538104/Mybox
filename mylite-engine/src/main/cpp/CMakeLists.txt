cmake_minimum_required(VERSION 3.22)
project(mybox_mylite LANGUAGES CXX ASM)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(MYLITE_CORE_SOURCES path_rules.cpp native_io.cpp native_install_guard.cpp syscall_gate.S)
add_library(mybox-mylite-core STATIC ${MYLITE_CORE_SOURCES})
set_target_properties(mybox-mylite-core PROPERTIES POSITION_INDEPENDENT_CODE ON)
target_compile_definitions(mybox-mylite-core PRIVATE _GNU_SOURCE)
target_compile_options(mybox-mylite-core PRIVATE -Wall -Wextra -Werror -fno-exceptions -fno-rtti)
if(ANDROID)
    if(NOT ANDROID_ABI STREQUAL "arm64-v8a" AND NOT ANDROID_ABI STREQUAL "x86_64")
        message(FATAL_ERROR "Native I/O supports arm64-v8a and x86_64 only")
    endif()
    add_library(mybox-mylite SHARED native_bridge.cpp native_uid.cpp)
    target_link_libraries(mybox-mylite PRIVATE mybox-mylite-core dl)
    target_link_options(mybox-mylite PRIVATE -Wl,-z,now -Wl,-z,relro -Wl,-z,max-page-size=16384)
else()
    enable_testing()
    add_library(mybox-mylite-test-core STATIC ${MYLITE_CORE_SOURCES})
    target_compile_definitions(mybox-mylite-test-core PRIVATE _GNU_SOURCE MYLITE_NATIVE_TESTING)
    target_compile_options(mybox-mylite-test-core PRIVATE -Wall -Wextra -Werror -fno-exceptions -fno-rtti)
    add_executable(mybox-mylite-native-test tests/native_core_test.cpp tests/wait_mask_test.cpp)
    target_compile_definitions(mybox-mylite-native-test PRIVATE MYLITE_NATIVE_TESTING)
    target_link_libraries(mybox-mylite-native-test PRIVATE mybox-mylite-test-core pthread)
    target_link_options(mybox-mylite-native-test PRIVATE -Wl,-z,now)
    add_test(NAME native-io COMMAND mybox-mylite-native-test)
    if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
        add_executable(mybox-mylite-benchmark tests/benchmark.cpp)
        target_compile_definitions(mybox-mylite-benchmark PRIVATE _GNU_SOURCE MYLITE_BENCH_BUILD_TYPE="$<CONFIG>")
        target_compile_options(mybox-mylite-benchmark PRIVATE -Wall -Wextra -Werror)
        target_link_libraries(mybox-mylite-benchmark PRIVATE mybox-mylite-core pthread)
        target_link_options(mybox-mylite-benchmark PRIVATE -Wl,-z,now)
    endif()
endif()
